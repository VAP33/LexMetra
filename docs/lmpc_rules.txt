applicability:

  - id: "APL-R3-CHAPTER-II-SCOPE"
    ruleReference: "Rule 3"
    description: |
      Chapter II does not apply to: (a) packages of commodities
      containing more than 25 kg or 25 litres, EXCLUDING cement and
      fertiliser sold in bags up to 50 kg; and (b) packaged commodities
      meant for industrial or institutional consumers.
    logic:
      anyOf:
        - allOf:
            - quantityExceeds: { value: 25, unit: "kg" }
            - not:
                allOf:
                  - commodityTypeMatches: ["cement", "fertiliser", "fertilizer"]
                  - quantityAtMost: { value: 50, unit: "kg" }
        - allOf:
            - quantityExceeds: { value: 25, unit: "l" }
            - not:
                allOf:
                  - commodityTypeMatches: ["cement", "fertiliser", "fertilizer"]
                  - quantityAtMost: { value: 50, unit: "kg" }
        - saleType: "industrial"
        - saleType: "institutional"
    outcomeWhenApplicable: APPLICABLE
    outcomeWhenNot: NOT_APPLICABLE

  - id: "APL-R26-A-SMALL-PACK"
    ruleReference: "Rule 26(a)"
    description: |
      Exemption for packages whose net weight or measure is 10 g or
      10 ml or less. For packages between 10 g and 20 g (10 ml and
      20 ml), only MRP and net quantity declarations are required.
    logic:
      anyOf:
        - quantityAtMost: { value: 10, unit: "g" }
        - quantityAtMost: { value: 10, unit: "ml" }
    outcome: EXEMPT
    partialRequirementBand:
      min: { value: 10, unit: "g_or_ml" }
      max: { value: 20, unit: "g_or_ml" }
      requiredFields: ["mrp", "netQuantityValue", "netQuantityUnit"]

  - id: "APL-R26-B-FAST-FOOD"
    ruleReference: "Rule 26(b)"
    description: |
      Packages containing fast food items packed by restaurant or hotel
      and the like are exempt.
    logic:
      allOf:
        - productCategory: "food"
        - packedByEstablishmentType: ["restaurant", "hotel", "quickService"]
    outcome: EXEMPT

  - id: "APL-R26-C-DPCO"
    ruleReference: "Rule 26(c)"
    description: |
      Scheduled and non-scheduled formulations covered under the
      Drugs (Price Control) Order, 1995 are exempt.
    logic:
      allOf:
        - productCategory: "drug"
        - coveredByDpco: true
    outcome: EXEMPT

  - id: "APL-R26-D-AGRI-BULK"
    ruleReference: "Rule 26(d)"
    description: |
      Agricultural farm produce in packages above 50 kg is exempt.
    logic:
      allOf:
        - commodityTypeMatches: ["agriculturalFarmProduce"]
        - quantityExceeds: { value: 50, unit: "kg" }
    outcome: EXEMPT

  - id: "R6-1-A-MANUFACTURER"
    ruleReference: "Rule 6(1)(a)"
    version: "2011.1"
    title: "Manufacturer / Packer / Importer name and full address"
    description: |
      Every package must bear the name and complete address of the
      manufacturer, or where the manufacturer is not the packer, both
      manufacturer and packer; imported packages must bear the importer.
      Explanation III to Rule 6(1)(a): for food packages this sub-rule
      does not apply; instead the PFA Act, 1954 applies.
    appliesWhen:
      saleType: ["retail", "ecommerce"]
      productCategory: "any"
    exemptWhen:
      anyOf:
        - applicabilityId: "APL-R26-A-SMALL-PACK"
        - productCategory: "food"
    requires:
      anyOf:
        - field: "manufacturer"
          companionField: "manufacturerAddress"
        - allOf:
            - field: "manufacturer"
            - field: "packer"
            - field: "packerAddress"
        - allOf:
            - field: "importer"
            - field: "importerAddress"
    validations:
      - id: "V-NAME-PRESENT"
        type: "presence"
        fields: ["manufacturer|packer|importer"]
        failAction: FAIL
      - id: "V-ADDRESS-PRESENT"
        type: "presence"
        fields: ["manufacturerAddress|packerAddress|importerAddress"]
        failAction: FAIL
      - id: "V-ADDRESS-COMPLETENESS"
        type: "compositeAddress"
        requiresComponents: ["cityOrDistrict", "stateOrUt"]
        optionalComponents: ["street", "premisesNumber", "pinCode"]
        onMissingCritical: UNCERTAIN
    severity: critical
    reviewPolicy: onUncertain
    notes: |
      Rule 10 Explanation defines "complete address" as the postal
      address of the factory, or street name, premises number, city,
      state and PIN code in other cases.

    ruleReference: "Rule 7(2)(i)"
    version: "2011.1"
    title: "PDP numeral height (weight or volume) — Table I"
    description: |
      Minimum height of numerals for net quantity declared in weight
      or volume. Table I: up to 200 g/ml → 1 mm normal / 2 mm blown;
      above 200 up to 500 → 2 mm / 4 mm; above 500 → 4 mm / 6 mm.
    appliesWhen:
      saleType: ["retail", "ecommerce"]
      netQuantityMeasuredBy: ["weight", "volume"]
    requires:
      field: "netQuantityNumeralHeightMm"
    validations:
      - id: "V-NUMERAL-HEIGHT-WEIGHT-VOLUME"
        type: "thresholdLookup"
        field: "netQuantityNumeralHeightMm"
        lookupBy: "netQuantityValue|netQuantityUnit"
        table:
          - whenQuantityAtMost: { value: 200, unit: "g_or_ml" }
            minHeightMm: { normal: 1, blownOrMolded: 2 }
          - whenQuantityBetween: { low: { value: 200, unit: "g_or_ml" }, high: { value: 500, unit: "g_or_ml" } }
            minHeightMm: { normal: 2, blownOrMolded: 4 }
          - whenQuantityAbove: { value: 500, unit: "g_or_ml" }
            minHeightMm: { normal: 4, blownOrMolded: 6 }
        failAction: FAIL
        uncertainIfMeasurementUncalibrated: true
    severity: major
    reviewPolicy: onUncertain

            minHeightMm: { normal: 6, blownOrMolded: 6 }

        uncertainIfMeasurementUncalibrated: true

    reviewPolicy: onUncertain
